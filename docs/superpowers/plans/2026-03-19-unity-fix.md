// --- NavMesh: bake at runtime using physics colliders ---
var ground = GameObject.Find("Ground");
if (ground != null)
{
    var surfaceType = RuntimeNavMeshBaker.ResolveSurfaceType();
    if (surfaceType != null)
    {
        ground.AddComponent(surfaceType);
    }
    ground.AddComponent<RuntimeNavMeshBaker>();
}

Note: `ResolveSurfaceType()` is a private static method. We need to make it internal or add a public accessor.

            // 3 * 20 / (18 * 100) = 0.033 per violation.
            // Threshold should be reachable within 2-3 violations.
            var rm = new UnityEngine.GameObject().AddComponent<ReportManager>();
            var field = typeof(ReportManager).GetField("globalSuspicionThreshold",
                System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);

            var value = (float)field.GetValue(rm);
            Assert.LessOrEqual(value, 0.05f,

            UnityEngine.Object.DestroyImmediate(rm.gameObject);
        }
    }
}

{
    if (RoleColors.TryGetValue(roleId, out var color))
    {
        var renderer = npc.GetComponent<Renderer>();
        if (renderer != null)
        {
            renderer.material = new Material(Shader.Find("Universal Render Pipeline/Lit"));
            renderer.material.color = color;
        }
    }
}
